class Api:
    def __init__(s, base, log): s.base = base; s.log = log
    def __call__(s, m, u, b=None, tok=None, raw=False, headers=None):
        data = json.dumps(b).encode() if b is not None else None
        req = urllib.request.Request(s.base + u, data=data, method=m, headers={"content-type": "application/json", **(headers or {})})
        if tok: req.add_header("authorization", f"Bearer {tok}")
        try:
            with urllib.request.urlopen(req, timeout=90) as r:
                d = r.read(); return r.status, (d.decode() if raw else (json.loads(d) if d else None)), dict(r.headers)
        except urllib.error.HTTPError as e:
            d = e.read()
            try: return e.code, json.loads(d), dict(e.headers)
            except Exception: return e.code, d.decode(), dict(e.headers)

"""Completion v2 — REAL Chromium journey against `python3 -m apps.main` (in-memory profile).

Journeys: A root→/app, register via dev token, sign in (cookie), orientation;
B run a real task → readable answer + facts; reload → session survives (cookie);

E Models runtime rows; F Capabilities panels; G error hint (bad template ref);

scenario save+replay → Promote (evidence refs, prompts auto-answered) → dashboard;
then Workbench run again → gold chip. Screenshots + facts → evidence/completion_v2/browser/.
"""

def free_port() -> int:
    s = socket.socket(); s.bind(("127.0.0.1", 0)); p = s.getsockname()[1]; s.close(); return p

def wait_http(url: str, timeout: float = 60) -> None:
    t0 = time.time()
    while time.time() - t0 < timeout:
        try:
            urllib.request.urlopen(url, timeout=2); return
        except Exception:
            time.sleep(0.4)
    raise RuntimeError("server did not come up")

def snap(page, name):
    page.screenshot(path=str(OUT / f"{name}.png"), full_page=True)

def register_and_login(page, base, email):
    page.goto(base + "/")
    page.wait_for_selector("#auth-view:not([hidden])")
    page.click("#tab-register")
    page.fill("#register-email", email); page.fill("#register-password", PW)
    page.click("#register-form button[type=submit]")
    page.wait_for_function("document.getElementById('verify-token').value.length > 10")
    token_prefilled = page.input_value("#verify-token")
    info = page.inner_text("#register-info")
    page.click("#verify-button")
    page.wait_for_function("document.getElementById('login-form').hidden === false", timeout=15000)
    page.fill("#login-email", email); page.fill("#login-password", PW)
    page.click("#login-form button[type=submit]")
    page.wait_for_selector("#main-view:not([hidden])")
    return token_prefilled, info

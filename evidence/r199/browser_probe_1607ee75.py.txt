def http(m,u,b=None,tok=None):
    req=urllib.request.Request(u,method=m,data=json.dumps(b).encode() if b is not None else None,headers={"Content-Type":"application/json",**({"Authorization":f"Bearer {tok}"} if tok else {})})
    try:
        with urllib.request.urlopen(req,timeout=5) as r: return r.status, json.loads(r.read() or b"{}")
    except urllib.error.HTTPError as e: return e.code, json.loads(e.read() or b"{}")
